use("laboratorio");

const oid  = { bsonType: "objectId" };
const str  = { bsonType: "string" };
const bool = { bsonType: "bool" };
const date = { bsonType: "date" };
const num  = { bsonType: ["int", "long", "double", "decimal"], minimum: 0 };
const en   = (...v) => ({ bsonType: "string", enum: v });
const nul  = (t) => ({ bsonType: [t.bsonType, "null"].flat() });

// Campos comunes de borrado lógico y auditoría de fechas
const borrado = {
  eliminado: bool, eliminado_por: nul(str), fecha_eliminacion: nul(date),
  fecha_creacion: date, fecha_actualizacion: nul(date),
};

function crear(nombre, required, properties, indices = []) {
  if (db.getCollectionNames().includes(nombre)) db[nombre].drop();
  db.createCollection(nombre, {
    validator: { $jsonSchema: { bsonType: "object", required, properties } },
    validationLevel: "strict", validationAction: "error",
  });
  indices.forEach(([keys, opts]) => db[nombre].createIndex(keys, opts || {}));
}

// 1. USUARIOS
crear("usuarios", ["nombre", "email", "password", "rol"], {
  nombre: str, email: str, password: str, // hash bcrypt
  rol: en("admin", "encargado", "usuario"),
  activo: bool, permiso_editar: bool, permiso_eliminar: bool, ...borrado,
}, [[{ email: 1 }, { unique: true }]]);

// 2. AUDITORÍAS (solo inserción)
crear("auditorias", ["entidad", "accion", "fecha_creacion"], {
  entidad: str, entidad_id: nul(str), accion: str,
  antes: {}, despues: {}, usuario: nul(str), ip: nul(str), fecha_creacion: date,
}, [[{ entidad: 1, entidad_id: 1 }], [{ fecha_creacion: -1 }]]);

// 3. SECUENCIAS  (_id = nombre, ej. "cotizaciones", "muestras-2026")
crear("secuencias", ["_id", "secuencia"], { _id: str, secuencia: { bsonType: ["int", "long"] } });

// 4. CLIENTES
crear("clientes", ["nombre", "tipo_documento", "numero_documento", "email", "tipo_cliente"], {
  nombre: str, tipo_documento: en("CC", "NIT", "CE", "TI"), numero_documento: str,
  email: str, telefono: str, direccion: str, tipo_cliente: en("interno", "externo"), ...borrado,
}, [[{ numero_documento: 1 }, { unique: true }]]);

// 5. SOLICITUDES
crear("solicitudes", ["cliente_id", "tipo_cliente", "descripcion", "estado"], {
  cliente_id: oid, tipo_cliente: en("interno", "externo"), descripcion: str,
  prioridad: en("baja", "media", "alta"), atencion_inmediata: bool,
  motivo_atencion_inmediata: str,
  estado: en("pendiente", "cotizada", "aceptada", "pagada", "rechazada"),
  requiere_pago: bool, ...borrado,
}, [[{ estado: 1, fecha_creacion: -1 }], [{ cliente_id: 1 }]]);

// 6. PARÁMETROS DE ANÁLISIS
crear("parametros_analisis", ["nombre", "descripcion", "precio"], {
  nombre: str, descripcion: str, precio: num, unidad: str, activo: bool, ...borrado,
}, [[{ nombre: 1 }, { unique: true }]]);

// 7. COTIZACIONES (items embebidos)
crear("cotizaciones", ["solicitud_id", "numero", "subtotal", "total", "estado", "items"], {
  solicitud_id: oid, numero: { bsonType: ["int", "long"] }, subtotal: num, total: num,
  estado: en("borrador", "enviada", "envio_fallido", "aceptada", "rechazada"),
  enviada_por_correo: bool, error_correo: str,
  items: { bsonType: "array", items: {
    bsonType: "object",
    required: ["parametro_id", "cantidad", "precio_unitario", "subtotal"],
    properties: {
      parametro_id: oid, descripcion: str,
      cantidad: { bsonType: ["int", "long"], minimum: 1 },
      precio_unitario: num, subtotal: num,
    } } },
  ...borrado,
}, [[{ numero: 1 }, { unique: true }], [{ solicitud_id: 1 }]]);

// 8. PAGOS (simulados)
crear("pagos", ["cotizacion_id", "solicitud_id", "referencia", "monto", "estado"], {
  cotizacion_id: oid, solicitud_id: oid, referencia: str, monto: num, metodo: str,
  estado: en("pendiente", "confirmado", "fallido"), fecha_confirmacion: nul(date), ...borrado,
}, [[{ referencia: 1 }, { unique: true }], [{ solicitud_id: 1 }],
    [{ cotizacion_id: 1 }], [{ estado: 1, fecha_creacion: -1 }]]);

// 9. MUESTRAS (parametros_ids reemplaza muestra_parametros)
crear("muestras", ["solicitud_id", "cliente_id", "nombre_muestra", "descripcion",
                   "tipo_fisico", "cantidad", "unidad", "estado_recepcion", "estado"], {
  solicitud_id: oid, cliente_id: oid, nombre_muestra: str, descripcion: str,
  tipo_fisico: en("solido", "liquido"), cantidad: num, unidad: en("g", "ml"),
  parametros_ids: { bsonType: "array", items: oid },
  verificacion_fisica: bool,
  estado_recepcion: en("pendiente", "aceptada", "rechazada"), motivo_rechazo: str,
  codigo: nul(str),               // '0042-2026'
  codigo_seguimiento: nul(str),   // aleatorio, 32 chars
  estado: en("ingresada", "en_proceso", "en_analisis", "resultados_validados",
             "cerrada", "rechazada", "en_almacen", "devuelta", "desechada"),
  rotulo_impreso: bool,
  rotulo_pdf_id: { bsonType: ["objectId", "null"] },   // GridFS
  ubicacion_clasificacion: { enum: ["ingresada", "en_proceso", "en_analisis", null] },
  ubicacion: str, ubicacion_actualizado_en: nul(date),
  fecha_recepcion: date, fecha_inicio: nul(date), fecha_estimada_entrega: nul(date),
  fecha_cierre: nul(date), fecha_limite_conservacion: nul(date),
  pendiente_disposicion: bool, ultimo_evento_tipo: nul(str), ultimo_evento_fecha: nul(date),
  ...borrado,
}, [
  [{ codigo: 1 }, { unique: true, partialFilterExpression: { codigo: { $type: "string" } } }],
  [{ codigo_seguimiento: 1 }, { unique: true, partialFilterExpression: { codigo_seguimiento: { $type: "string" } } }],
  [{ solicitud_id: 1 }], [{ cliente_id: 1 }],
  [{ estado: 1, fecha_recepcion: -1 }], [{ pendiente_disposicion: 1 }],
]);
